From 04-UI-SPEC.md — focus trap pattern:
```typescript
useEffect(() => {
  if (!show) return;
  const focusable = modalRef.current?.querySelectorAll('button, [href], input, [tabindex]:not([tabindex="-1"])') as NodeListOf<HTMLElement>;
  const first = focusable?.[0]; const last = focusable?.[focusable.length - 1];
  const trap = (e: KeyboardEvent) => {
    if (e.key === 'Escape') { dismiss(); return; }
    if (e.key !== 'Tab') return;
    if (e.shiftKey) { if (document.activeElement === first) { e.preventDefault(); last?.focus(); } }
    else { if (document.activeElement === last) { e.preventDefault(); first?.focus(); } }
  };
  const previousFocus = document.activeElement as HTMLElement;
  window.addEventListener('keydown', trap);
  first?.focus();
  return () => { window.removeEventListener('keydown', trap); previousFocus?.focus(); };
}, [show]);
```
